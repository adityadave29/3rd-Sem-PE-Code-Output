-- ============================================
-- SETUP
-- ============================================
DROP TABLE IF EXISTS customers;


CREATE TABLE customers (
   customer_id   INT PRIMARY KEY,
   customer_name VARCHAR(50),
   owner_user    NAME NOT NULL   -- maps each row to a DB role allowed to see it
);


INSERT INTO customers (customer_id, customer_name, owner_user) VALUES
   (801, 'Edward Smith',  'app_user_1'),
   (802, 'Alice Johnson', 'app_user_2'),
   (803, 'Michael Brown', 'app_user_1');


-- ============================================
-- CREATE RESTRICTED ROLES
-- ============================================
DROP ROLE IF EXISTS app_user_1;
DROP ROLE IF EXISTS app_user_2;
DROP ROLE IF EXISTS app_role;


CREATE ROLE app_role NOLOGIN;
CREATE ROLE app_user_1 NOLOGIN IN ROLE app_role;
CREATE ROLE app_user_2 NOLOGIN IN ROLE app_role;


GRANT SELECT ON customers TO app_role;   -- role CAN query the table


-- ============================================
-- ENABLE RLS AND CREATE POLICY
-- Rule: a role can only see rows where owner_user = current_user
-- ============================================
ALTER TABLE customers ENABLE ROW LEVEL SECURITY;
ALTER TABLE customers FORCE ROW LEVEL SECURITY;  -- applies even to table owner


DROP POLICY IF EXISTS customer_filter_policy ON customers;


CREATE POLICY customer_filter_policy
   ON customers
   FOR SELECT
   USING (owner_user = current_user);


-- ============================================
-- TEST 1: What app_user_1 can see directly (RLS in effect)
-- ============================================
\echo '--- Direct SELECT as app_user_1 (RLS filtering applied) ---'


SET ROLE app_user_1;


SELECT * FROM customers;   -- should only return rows 801 and 803, NOT 802


RESET ROLE;
-- Expected: only customer_id 801 and 803 visible. 802 (Alice, owned by app_user_2) is invisible.




-- ============================================
-- TEST 2: Attack against a row app_user_1 is NOT allowed to see (customer_id 802)
-- ============================================
\echo '--- Attack as app_user_1 against customer_id 802 (not owned by app_user_1) ---'


SET ROLE app_user_1;


DO $$
DECLARE
   dummy INT;
BEGIN
   SELECT 1 / (
       CASE SUBSTRING(customer_name, 1, 1) WHEN 'A' THEN 0 ELSE 1 END
   )
   INTO dummy
   FROM customers
   WHERE customer_id = 802;


   RAISE NOTICE 'No error - RLS filtered the row before the CASE/divide ever evaluated. dummy = %', dummy;
EXCEPTION
   WHEN division_by_zero THEN
       RAISE NOTICE 'CAUGHT ERROR: division by zero';
END $$;


RESET ROLE;
-- Expected: NO error, NO rows match -> SELECT INTO returns NULL with a "no rows" style outcome,
-- no exception is raised, regardless of the guessed character.
-- RLS removes the row before the CASE/divide logic is ever evaluated against it.




-- ============================================
-- TEST 3: Attack against a row app_user_1 IS allowed to see (customer_id 801)
-- ============================================
\echo '--- Attack as app_user_1 against customer_id 801 (owned by app_user_1) ---'


SET ROLE app_user_1;


DO $$
DECLARE
   dummy INT;
BEGIN
   SELECT 1 / (
       CASE SUBSTRING(customer_name, 1, 1) WHEN 'E' THEN 0 ELSE 1 END
   )
   INTO dummy
   FROM customers
   WHERE customer_id = 801;


   RAISE NOTICE 'No error. dummy = %', dummy;
EXCEPTION
   WHEN division_by_zero THEN
       RAISE NOTICE 'CAUGHT ERROR: division by zero -- inference succeeds, row was already visible to this user anyway';
END $$;


RESET ROLE;
-- Expected: division_by_zero exception fires, same as before RLS,
-- because this row passes the RLS filter and reaches evaluation.
